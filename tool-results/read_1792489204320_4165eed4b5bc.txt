     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import { motion, AnimatePresence, useReducedMotion } from "framer-motion";
     5→import { ArrowLeft, ArrowRight, Check, ChevronRight } from "lucide-react";
     6→import { useForm, FormProvider } from "react-hook-form";
     7→import { zodResolver } from "@hookform/resolvers/zod";
     8→import { Button } from "@/components/ui/button";
     9→import { Input } from "@/components/ui/input";
    10→import { Textarea } from "@/components/ui/textarea";
    11→import { Label } from "@/components/ui/label";
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from "@/components/ui/select";
    19→import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
    20→import { Checkbox } from "@/components/ui/checkbox";
    21→import { Link } from "@/lib/router";
    22→import { useServices, useContentStore } from "@/lib/content-store";
    23→import {
    24→  inquirySchema,
    25→  step1Schema,
    26→  step2Schema,
    27→  step3Schema,
    28→  step4Schema,
    29→  DEFAULT_INQUIRY_VALUES,
    30→  PROJECT_TYPES,
    31→  BUDGET_TIERS,
    32→  TIMELINES,
    33→  CONTACT_METHODS,
    34→  toInquiry,
    35→  type InquiryFormValues,
    36→} from "@/lib/inquiry-schema";
    37→import { toast } from "sonner";
    38→
    39→const STEPS = [
    40→  { id: 1, label: "About you", schema: step1Schema },
    41→  { id: 2, label: "Project", schema: step2Schema },
    42→  { id: 3, label: "Details", schema: step3Schema },
    43→  { id: 4, label: "Review & consent", schema: step4Schema },
    44→];
    45→
    46→interface ConsultationWizardProps {
    47→  /** Optional prefill bundle (e.g. from Solution Configurator) */
    48→  initial?: Partial<InquiryFormValues>;
    49→  /** On submit-success callback (e.g. parent wants to reset/clear) */
    50→  onSubmitted?: () => void;
    51→}
    52→
    53→export function ConsultationWizard({ initial, onSubmitted }: ConsultationWizardProps) {
    54→  const reduce = useReducedMotion();
    55→  const services = useServices();
    56→  const { upsertInquiry } = useContentStore();
    57→  const [step, setStep] = useState(1);
    58→  const [submitting, setSubmitting] = useState(false);
    59→  const [submitted, setSubmitted] = useState(false);
    60→  const [autoResetTimer, setAutoResetTimer] = useState<ReturnType<typeof setTimeout> | null>(null);
    61→
    62→  const form = useForm<InquiryFormValues>({
    63→    resolver: zodResolver(inquirySchema),
    64→    defaultValues: { ...DEFAULT_INQUIRY_VALUES, ...initial },
    65→    mode: "onChange",
    66→  });
    67→
    68→  // Prefill if initial changes (e.g. user changes bundle in configurator)
    69→  useEffect(() => {
    70→    if (initial) {
    71→      Object.entries(initial).forEach(([k, v]) => {
    72→        form.setValue(k as keyof InquiryFormValues, v as never, {
    73→          shouldValidate: false,
    74→          shouldDirty: true,
    75→        });
    76→      });
    77→    }
    78→    // eslint-disable-next-line react-hooks/exhaustive-deps
    79→  }, [JSON.stringify(initial)]);
    80→
    81→  // Clear auto-reset on unmount
    82→  useEffect(() => {
    83→    return () => {
    84→      if (autoResetTimer) clearTimeout(autoResetTimer);
    85→    };
    86→  }, [autoResetTimer]);
    87→
    88→  // Field-level schema for the current step
    89→  const stepSchema = STEPS[step - 1].schema;
    90→  const stepFields = useMemo(() => {
    91→    const keys = stepSchema && typeof stepSchema === "object" && "shape" in stepSchema
    92→      ? Object.keys((stepSchema as { shape: Record<string, unknown> }).shape)
    93→      : [];
    94→    return keys as (keyof InquiryFormValues)[];
    95→  }, [stepSchema]);
    96→
    97→  const validateStep = async (): Promise<boolean> => {
    98→    const values = form.getValues();
    99→    const result = await stepSchema.safeParseAsync(
   100→      Object.fromEntries(stepFields.map((k) => [k, values[k]]))
   101→    );
   102→    if (!result.success) {
   103→      // Apply errors to the form
   104→      stepFields.forEach((field) => {
   105→        const err = result.error.issues.find((i) => i.path[0] === field);
   106→        if (err) {
   107→          form.setError(field, { type: "manual", message: err.message });
   108→        } else {
   109→          form.clearErrors(field);
   110→        }
   111→      });
   112→      // Trigger validation display
   113→      await form.trigger(stepFields);
   114→      return false;
   115→    }
   116→    return true;
   117→  };
   118→
   119→  const next = async () => {
   120→    const ok = await validateStep();
   121→    if (ok && step < STEPS.length) {
   122→      setStep((s) => s + 1);
   123→      if (typeof window !== "undefined") {
   124→        window.scrollTo({ top: 0, behavior: "smooth" });
   125→      }
   126→    }
   127→  };
   128→
   129→  const back = () => {
   130→    if (step > 1) setStep((s) => s - 1);
   131→  };
   132→
   133→  const onSubmit = form.handleSubmit(async (values) => {
   134→    setSubmitting(true);
   135→    // Simulate 800ms network latency
   136→    await new Promise((r) => setTimeout(r, 800));
   137→    const inquiry = toInquiry(values);
   138→    upsertInquiry(inquiry);
   139→    setSubmitting(false);
   140→    setSubmitted(true);
   141→    toast.success("Inquiry received (demo).", {
   142→      description: "We'll be in touch within 1 business day.",
   143→    });
   144→    onSubmitted?.();
   145→    const t = setTimeout(() => {
   146→      setSubmitted(false);
   147→      setStep(1);
   148→      form.reset({ ...DEFAULT_INQUIRY_VALUES });
   149→    }, 6000);
   150→    setAutoResetTimer(t);
   151→  });
   152→
   153→  const submitAnother = () => {
   154→    if (autoResetTimer) clearTimeout(autoResetTimer);
   155→    setSubmitted(false);
   156→    setStep(1);
   157→    form.reset({ ...DEFAULT_INQUIRY_VALUES });
   158→  };
   159→
   160→  const progress = (step / STEPS.length) * 100;
   161→  const values = form.watch();
   162→
   163→  if (submitted) {
   164→    return (
   165→      <div
   166→        className="rounded-xl border p-8 text-center"
   167→        style={{
   168→          borderColor: "color-mix(in srgb, var(--success) 35%, transparent)",
   169→          backgroundColor: "color-mix(in srgb, var(--success) 8%, var(--surface-1))",
   170→        }}
   171→        role="status"
   172→        aria-live="polite"
   173→      >
   174→        <motion.div
   175→          initial={reduce ? false : { opacity: 0, scale: 0.9 }}
   176→          animate={{ opacity: 1, scale: 1 }}
   177→          transition={{ duration: 0.3 }}
   178→          className="mx-auto flex size-14 items-center justify-center rounded-full border"
   179→          style={{
   180→            borderColor: "var(--success)",
   181→            backgroundColor: "color-mix(in srgb, var(--success) 14%, transparent)",
   182→          }}
   183→        >
   184→          <Check className="size-7" style={{ color: "var(--success)" }} aria-hidden />
   185→        </motion.div>
   186→        <h3
   187→          className="mt-5 font-display text-xl font-semibold"
   188→          style={{ color: "var(--text-primary)" }}
   189→        >
   190→          Thanks — your inquiry has been received (demo).
   191→        </h3>
   192→        <p
   193→          className="mt-2 text-sm"
   194→          style={{ color: "var(--text-secondary)" }}
   195→        >
   196→          We&apos;ll be in touch within 1 business day. For urgent matters,
   197→          reach us at the contact info shown on this page.
   198→        </p>
   199→        <p
   200→          className="mt-2 text-xs"
   201→          style={{ color: "var(--text-muted)" }}
   202→        >
   203→          Inquiry ID: <span className="font-mono">{values.email ? values.email.slice(0, 4) : ""}-demo</span>
   204→        </p>
   205→        <div className="mt-6 flex flex-wrap items-center justify-center gap-3">
   206→          <Button
   207→            type="button"
   208→            onClick={submitAnother}
   209→            variant="outline"
   210→            style={{ borderColor: "var(--border-strong)" }}
   211→          >
   212→            Submit another
   213→          </Button>
   214→          <Button asChild variant="ghost" style={{ color: "var(--text-muted)" }}>
   215→            <Link to="/">Back to home</Link>
   216→          </Button>
   217→        </div>
   218→      </div>
   219→    );
   220→  }
   221→
   222→  return (
   223→    <FormProvider {...form}>
   224→      <form onSubmit={onSubmit} className="space-y-6" noValidate>
   225→        {/* Stepper header */}
   226→        <div
   227→          className="rounded-xl border p-5"
   228→          style={{
   229→            borderColor: "var(--border-subtle)",
   230→            backgroundColor: "var(--surface-1)",
   231→          }}
   232→        >
   233→          <div className="flex items-center justify-between">
   234→            <div className="flex items-center gap-2">
   235→              {STEPS.map((s, i) => {
   236→                const isActive = step === s.id;
   237→                const isComplete = step > s.id;
   238→                return (
   239→                  <div key={s.id} className="flex items-center">
   240→                    <button
   241→                      type="button"
   242→                      onClick={() => isComplete && setStep(s.id)}
   243→                      disabled={!isComplete}
   244→                      aria-current={isActive ? "step" : undefined}
   245→                      className="inline-flex items-center gap-2 text-xs"
   246→                      style={{
   247→                        color: isActive || isComplete
   248→                          ? "var(--accent-bright)"
   249→                          : "var(--text-muted)",
   250→                        cursor: isComplete ? "pointer" : "default",
   251→                      }}
   252→                    >
   253→                      <span
   254→                        className="flex size-6 items-center justify-center rounded-full border font-mono text-[10px]"
   255→                        style={{
   256→                          borderColor: isActive || isComplete
   257→                            ? "var(--accent)"
   258→                            : "var(--border-strong)",
   259→                          backgroundColor: isActive
   260→                            ? "color-mix(in srgb, var(--accent) 14%, transparent)"
   261→                            : "transparent",
   262→                          color: isActive || isComplete
   263→                            ? "var(--accent-bright)"
   264→                            : "var(--text-muted)",
   265→                        }}
   266→                      >
   267→                        {isComplete ? <Check className="size-3" aria-hidden /> : s.id}
   268→                      </span>
   269→                      <span className="hidden sm:inline">{s.label}</span>
   270→                    </button>
   271→                    {i < STEPS.length - 1 && (
   272→                      <ChevronRight
   273→                        className="mx-1 size-3"
   274→                        style={{ color: "var(--border-strong)" }}
   275→                        aria-hidden
   276→                      />
   277→                    )}
   278→                  </div>
   279→                );
   280→              })}
   281→            </div>
   282→            <div className="text-[10px] font-mono uppercase tracking-widest" style={{ color: "var(--text-muted)" }}>
   283→              {step} / {STEPS.length}
   284→            </div>
   285→          </div>
   286→          {/* Progress bar */}
   287→          <div
   288→            className="mt-4 h-1 w-full overflow-hidden rounded-full"
   289→            style={{ backgroundColor: "var(--surface-2)" }}
   290→            aria-hidden
   291→          >
   292→            <motion.div
   293→              className="h-full"
   294→              animate={{ width: `${progress}%` }}
   295→              transition={{ duration: 0.4, ease: [0.16, 1, 0.3, 1] }}
   296→              style={{
   297→                background: "linear-gradient(90deg, var(--accent), var(--accent-cyan))",
   298→              }}
   299→            />
   300→          </div>
   301→        </div>
   302→
   303→        {/* Steps */}
   304→        <AnimatePresence mode="wait">
   305→          <motion.div
   306→            key={step}
   307→            initial={reduce ? false : { opacity: 0, y: 12 }}
   308→            animate={{ opacity: 1, y: 0 }}
   309→            exit={reduce ? undefined : { opacity: 0, y: -12 }}
   310→            transition={{ duration: 0.25 }}
   311→          >
   312→            {step === 1 && <Step1 form={form} />}
   313→            {step === 2 && <Step2 form={form} services={services} />}
   314→            {step === 3 && <Step3 form={form} />}
   315→            {step === 4 && <Step4 form={form} values={values} services={services} />}
   316→          </motion.div>
   317→        </AnimatePresence>
   318→
   319→        {/* Nav buttons */}
   320→        <div className="flex items-center justify-between gap-3">
   321→          <Button
   322→            type="button"
   323→            variant="outline"
   324→            onClick={back}
   325→            disabled={step === 1}
   326→            className="inline-flex items-center gap-2"
   327→            style={{ borderColor: "var(--border-strong)" }}
   328→          >
   329→            <ArrowLeft className="size-4" aria-hidden />
   330→            Back
   331→          </Button>
   332→          {step < STEPS.length ? (
   333→            <Button
   334→              type="button"
   335→              onClick={next}
   336→              className="inline-flex items-center gap-2"
   337→              style={{ backgroundColor: "var(--accent)", color: "#fff" }}
   338→            >
   339→              Continue
   340→              <ArrowRight className="size-4" aria-hidden />
   341→            </Button>
   342→          ) : (
   343→            <Button
   344→              type="submit"
   345→              disabled={submitting}
   346→              className="inline-flex items-center gap-2"
   347→              style={{ backgroundColor: "var(--accent)", color: "#fff" }}
   348→            >
   349→              {submitting ? (
   350→                <>
   351→                  <span
   352→                    className="inline-flex size-4 animate-spin rounded-full border-2 border-white/40 border-t-white"
   353→                    aria-hidden
   354→                  />
   355→                  Submitting…
   356→                </>
   357→              ) : (
   358→                <>
   359→                  <Check className="size-4" aria-hidden />
   360→                  Submit inquiry
   361→                </>
   362→              )}
   363→            </Button>
   364→          )}
   365→        </div>
   366→      </form>
   367→    </FormProvider>
   368→  );
   369→}

   578→// ============================================================
   579→// STEP 4 — REVIEW & CONSENT
   580→// ============================================================
   581→function Step4({
   582→  form,
   583→  values,
   584→  services,
   585→}: {
   586→  form: ReturnType<typeof useForm<InquiryFormValues>>;
   587→  values: InquiryFormValues;
   588→  services: ReturnType<typeof useServices>;
   589→}) {
   590→  const selectedService = services.find((s) => s.slug === values.serviceSlug);
   591→
   592→  return (

   687→function Field({
   688→  label,
   689→  required,
   690→  error,
   691→  hint,
   692→  children,
   693→}: {
   694→  label: string;
   695→  required?: boolean;
   696→  error?: string;
   697→  hint?: string;
   698→  children: React.ReactNode;
   699→}) {
   700→  return (
   701→    <div className="space-y-1.5">
   702→      <div className="flex items-center justify-between">
   703→        <Label className="text-xs font-medium uppercase tracking-widest" style={{ color: "var(--text-muted)" }}>
   704→          {label}
   705→          {required && <span style={{ color: "var(--accent-bright)" }}> *</span>}
   706→        </Label>
   707→        {hint && <span className="text-[10px] font-mono" style={{ color: "var(--text-muted)" }}>{hint}</span>}
   708→      </div>
   709→      {children}
   710→      {error && (
   711→        <p
   712→          role="alert"
   713→          aria-live="polite"
   714→          className="text-xs"
   715→          style={{ color: "var(--danger)" }}
   716→        >
   717→          {error}
   718→        </p>
   719→      )}
   720→    </div>
   721→  );
   722→}
   723→
   724→function ReviewRow({ label, value }: { label: string; value?: string }) {
   725→  if (!value) return null;
   726→  return (
   727→    <div className="flex items-start justify-between gap-3">
   728→      <span className="text-[10px] uppercase tracking-widest" style={{ color: "var(--text-muted)" }}>
   729→        {label}
   730→      </span>
   731→      <span className="text-right text-xs" style={{ color: "var(--text-secondary)" }}>
   732→        {value}
   733→      </span>
   734→    </div>
   735→  );
   736→}

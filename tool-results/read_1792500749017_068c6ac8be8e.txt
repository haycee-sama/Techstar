   110→// Deep clone helper (state is plain JSON-safe)
   111→function clone<T>(value: T): T {
   112→  if (typeof structuredClone === "function") return structuredClone(value);
   113→  return JSON.parse(JSON.stringify(value));
   114→}

   116→const DEFAULT_COOKIE_PREFS: CookiePreferences = {
   117→  essential: true,
   118→  analytics: false,
   119→  marketing: false,
   120→  functional: true,

   123→// ============================================================
   124→// PROVIDER
   125→// ============================================================
   126→export function ContentStoreProvider({ children }: { children: ReactNode }) {
   127→  const [state, setState] = useState<ContentState>(() => clone(seedData));
   128→  const [isDirty, setIsDirty] = useState(false);
   129→  const [lastSavedAt, setLastSavedAt] = useState<string | null>(null);
   130→  const [hydrated, setHydrated] = useState(false);
   131→  const [cookiePrefs, setCookiePrefsState] = useState<CookiePreferences>(DEFAULT_COOKIE_PREFS);
   132→  const saveTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   133→
   134→  // ---- Hydrate from localStorage on mount ----
   135→  useEffect(() => {
   136→    try {
   137→      const raw = window.localStorage.getItem(STORAGE_KEY);
   138→      if (raw) {
   139→        const parsed = JSON.parse(raw) as ContentState;
   140→        // Light sanity check
   141→        if (parsed && parsed.services && parsed.brand) {
   142→          setState(parsed);
   143→        }
   144→      }
   145→      const cookieRaw = window.localStorage.getItem(COOKIE_KEY);
   146→      if (cookieRaw) {
   147→        setCookiePrefsState(JSON.parse(cookieRaw) as CookiePreferences);
   148→      }
   149→    } catch {
   150→      // fall back to seed
   151→    }
   152→    setHydrated(true);
   153→  }, []);
   154→
   155→  // ---- Persist (debounced) ----
   156→  useEffect(() => {
   157→    if (!hydrated) return;
   158→    setIsDirty(true);
   159→    if (saveTimer.current) clearTimeout(saveTimer.current);
   160→    saveTimer.current = setTimeout(() => {
   161→      try {
   162→        window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
   163→        setLastSavedAt(new Date().toISOString());
   164→        setIsDirty(false);
   165→      } catch {
   166→        // quota / unavailable
   167→      }
   168→    }, SAVE_DEBOUNCE_MS);
   169→    return () => {
   170→      if (saveTimer.current) clearTimeout(saveTimer.current);
   171→    };
   172→  }, [state, hydrated]);
   173→
   174→  // ---- Cookie prefs persistence ----
   175→  useEffect(() => {
   176→    if (!hydrated) return;
   177→    try {
   178→      window.localStorage.setItem(COOKIE_KEY, JSON.stringify(cookiePrefs));
   179→    } catch {
   180→      // ignore
   181→    }
   182→  }, [cookiePrefs, hydrated]);
   183→
   184→  // ============================================================
   185→  // READ LOOKUPS (memoized)
   186→  // ============================================================
   187→  const getService = useCallback(
   188→    (slug: string) => state.services.find((s) => s.slug === slug),
   189→    [state.services]
   190→  );
   191→  const getServices = useCallback(() => state.services, [state.services]);
   192→  const getProject = useCallback(
   193→    (slug: string) => state.projects.find((p) => p.slug === slug),
   194→    [state.projects]
   195→  );
   196→  const getArticle = useCallback(
   197→    (slug: string) => state.articles.find((a) => a.slug === slug),
   198→    [state.articles]
   199→  );
   200→  const getAnnouncement = useCallback(
   201→    () => state.announcements.find((a) => a.active) || undefined,
   202→    [state.announcements]
   203→  );
   204→  const getBrand = useCallback(() => state.brand, [state.brand]);
   205→  const getGlobalSettings = useCallback(() => state.globalSettings, [state.globalSettings]);
   206→  const getInquiries = useCallback(() => state.inquiries, [state.inquiries]);
   207→  const getMedia = useCallback(() => state.media, [state.media]);
   208→  const getPage = useCallback(
   209→    (slug: string) => state.pages.find((p) => p.slug === slug),
   210→    [state.pages]
   211→  );

   213→  // ============================================================
   214→  // MUTATIONS
   215→  // ============================================================
   216→  const updateBrand = useCallback((partial: Partial<BrandSettings>) => {

   218→  }, []);
   219→
   220→  const updateService = useCallback((slug: string, partial: Partial<Service>) => {

   222→      ...prev,
   223→      services: prev.services.map((s) => (s.slug === slug ? { ...s, ...partial } : s)),

   251→  }, []);

   316→

   318→    setState((prev) => ({
   319→      ...prev,

   328→        ...prev,
   329→        media: exists

   331→          : [media, ...prev.media],
   332→      };
   333→    });
   334→  }, []);
   335→
   336→  const deleteMedia = useCallback((id: string) => {
   337→    setState((prev) => ({
   338→      ...prev,
   339→      media: prev.media.filter((m) => m.id !== id),
   340→    }));
   341→  }, []);
   342→
   343→  const upsertPage = useCallback((page: PageContent) => {
   344→    setState((prev) => {
   345→      const exists = prev.pages.some((p) => p.slug === page.slug);
   346→      return {
   347→        ...prev,
   348→        pages: exists
   349→          ? prev.pages.map((p) => (p.slug === page.slug ? page : p))
   350→          : [...prev.pages, page],
   351→      };
   352→    });
   353→  }, []);
   354→
   355→  const deletePage = useCallback((slug: string) => {
   356→    setState((prev) => ({
   357→      ...prev,
   358→      pages: prev.pages.filter((p) => p.slug !== slug),
   359→    }));
   360→  }, []);
   361→
   362→  const updatePage = useCallback((slug: string, partial: Partial<PageContent>) => {
   363→    setState((prev) => ({
   364→      ...prev,
   365→      pages: prev.pages.map((p) =>
   366→        p.slug === slug
   367→          ? { ...p, ...partial, updatedAt: new Date().toISOString() }
   368→          : p
   369→      ),
   370→    }));
   371→  }, []);
   372→
   373→  const updatePageSections = useCallback(
   374→    (slug: string, sections: PageContent["sections"]) => {
   375→      setState((prev) => ({
   376→        ...prev,
   377→        pages: prev.pages.map((p) =>
   378→          p.slug === slug
   379→            ? { ...p, sections, updatedAt: new Date().toISOString() }
   380→            : p
   381→        ),
   382→      }));
   383→    },
   384→    []
   385→  );
   386→
   387→  const publishPage = useCallback((slug: string) => {
   388→    const stamp = new Date().toISOString();
   389→    setState((prev) => ({
   390→      ...prev,
   391→      pages: prev.pages.map((p) => {
   392→        if (p.slug !== slug) return p;
   393→        const snapshot: PublishedPageSnapshot = {
   394→          title: p.title,
   395→          excerpt: p.excerpt,
   396→          bodyHtml: p.bodyHtml,
   397→          featuredImage: p.featuredImage,
   398→          authorId: p.authorId,
   399→          seo: p.seo ? { ...p.seo } : { title: p.title },
   400→          publishedAt: stamp,
   401→        };
   402→        return {
   403→          ...p,
   404→          status: "published" as const,
   405→          publishedAt: stamp,
   406→          updatedAt: stamp,
   407→          publishedSnapshot: snapshot,
   408→          // Phase 4 visual builder — capture the working sections as the
   409→          // last-published snapshot so the public preview route renders
   410→          // a stable version even as the user keeps editing the draft.
   411→          publishedSections: p.sections ? p.sections.map((s) => ({ ...s })) : [],
   412→        };
   413→      }),
   414→    }));
   415→  }, []);
   416→
   417→  const unpublishPage = useCallback((slug: string) => {
   418→    const stamp = new Date().toISOString();
   419→    setState((prev) => ({
   420→      ...prev,
   421→      pages: prev.pages.map((p) =>
   422→        p.slug === slug
   423→          ? { ...p, status: "archived" as const, updatedAt: stamp, publishedSections: [] }
   424→          : p
   425→      ),
   426→    }));
   427→  }, []);
   428→
   429→  const updateAnnouncement = useCallback(
   430→    (id: string, partial: Partial<Announcement>) => {
   431→      setState((prev) => ({
   432→        ...prev,
   433→        announcements: prev.announcements.map((a) =>
   434→          a.id === id ? { ...a, ...partial } : a
   435→        ),
   436→      }));
   437→    },
   438→    []
   439→  );
   440→
   441→  const upsertAnnouncement = useCallback((a: Announcement) => {
   442→    setState((prev) => {
   443→      const exists = prev.announcements.some((x) => x.id === a.id);

   445→      // deactivate all others. Otherwise leave active flags alone.
   446→      const announcements = exists
   447→        ? prev.announcements.map((x) =>
   448→            x.id === a.id
   449→              ? a
   450→              : a.active
   451→              ? { ...x, active: false }
   452→              : x
   453→          )
   454→        : a.active
   455→        ? [...prev.announcements.map((x) => ({ ...x, active: false })), a]
   456→        : [...prev.announcements, a];
   457→      return { ...prev, announcements };
   458→    });
   459→  }, []);
   460→
   461→  const deleteAnnouncement = useCallback((id: string) => {
   462→    setState((prev) => ({
   463→      ...prev,
   464→      announcements: prev.announcements.filter((a) => a.id !== id),
   465→    }));
   466→  }, []);
   467→

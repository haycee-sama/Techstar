   161→function BrandTab() {
   162→  const brand = useBrand();
   163→  const { updateBrand } = useContentStore();
   164→  const [draft, setDraft] = useState<BrandSettings>(brand);
   165→
   166→  useEffect(() => setDraft(brand), [brand]);
   167→
   168→  // Apply CSS variables live as the user edits colors
   169→  useEffect(() => {
   170→    if (typeof document === "undefined") return;
   171→    const root = document.documentElement;
   172→    root.style.setProperty("--accent", draft.colors.accent);
   173→    root.style.setProperty("--accent-cyan", draft.colors.accentCyan);
   174→    root.style.setProperty("--accent-bright", draft.colors.accentBright ?? draft.colors.accent);
   175→    if (draft.colors.bgDeep) root.style.setProperty("--bg-deep", draft.colors.bgDeep);
   176→    if (draft.colors.bgNavy) root.style.setProperty("--bg-navy", draft.colors.bgNavy);
   177→  }, [draft.colors]);
   178→
   179→  function save() {
   180→    updateBrand(draft);
   181→    toast.success("Brand settings saved.");
   182→  }
   183→
   184→  function setTypography(preset: string) {
   185→    const [display, body] = preset.split("+");
   186→    setDraft({ ...draft, typography: { display, body } });
   187→  }
   188→
   189→  return (
   190→    <Card>
   191→      <CardHeader>
   192→        <CardTitle className="font-display">Brand</CardTitle>
   193→        <CardDescription>
   194→          Company name, tagline, logo treatment, brand colors, and
   195→          typography. Color changes apply live to this admin screen
   196→          and to the public site on next visit.
   197→        </CardDescription>
   198→      </CardHeader>
   199→      <CardContent className="space-y-4">
   200→        <div className="grid gap-3 sm:grid-cols-2">
   201→          <div className="space-y-1.5">
   202→            <Label htmlFor="brand-name" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   203→              Company name
   204→            </Label>
   205→            <Input
   206→              id="brand-name"
   207→              value={draft.companyName}
   208→              onChange={(e) => setDraft({ ...draft, companyName: e.target.value })}
   209→            />
   210→          </div>
   211→          <div className="space-y-1.5">
   212→            <Label htmlFor="brand-tagline" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   213→              Tagline
   214→            </Label>
   215→            <Input
   216→              id="brand-tagline"
   217→              value={draft.tagline}
   218→              onChange={(e) => setDraft({ ...draft, tagline: e.target.value })}
   219→            />
   220→          </div>
   221→        </div>
   222→
   223→        <div className="space-y-1.5">
   224→          <Label htmlFor="brand-desc" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   225→            Short description
   226→          </Label>
   227→          <Textarea
   228→            id="brand-desc"
   229→            value={draft.shortDescription}
   230→            onChange={(e) => setDraft({ ...draft, shortDescription: e.target.value })}
   231→            rows={3}
   232→          />
   233→        </div>
   234→
   235→        <div className="space-y-1.5">
   236→          <Label className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   237→            Logo variant
   238→          </Label>
   239→          <Select
   240→            value={draft.logoVariant === "default" ? "full" : draft.logoVariant}
   241→            onValueChange={(v) => setDraft({ ...draft, logoVariant: v as BrandSettings["logoVariant"] })}
   242→          >
   243→            <SelectTrigger>
   244→              <SelectValue />

   333→function ContactTab() {
   334→  const gs = useGlobalSettings();
   335→  const { updateGlobalSettings } = useContentStore();
   336→  const [draft, setDraft] = useState<GlobalSettings>(gs);
   337→
   338→  useEffect(() => setDraft(gs), [gs]);
   339→
   340→  function save() {
   341→    updateGlobalSettings(draft);
   342→    toast.success("Contact settings saved.");
   343→  }
   344→
   345→  function updateHours(i: number, next: Partial<{ day: string; hours: string }>) {
   346→    const hours = draft.contact.hours.slice();
   347→    hours[i] = { ...hours[i], ...next };
   348→    setDraft({ ...draft, contact: { ...draft.contact, hours } });
   349→  }
   350→
   351→  function addHour() {
   352→    setDraft({
   353→      ...draft,
   354→      contact: { ...draft.contact, hours: [...draft.contact.hours, { day: "", hours: "" }] },
   355→    });
   356→  }
   357→
   358→  function removeHour(i: number) {
   359→    setDraft({
   360→      ...draft,
   361→      contact: {
   362→        ...draft.contact,
   363→        hours: draft.contact.hours.filter((_, idx) => idx !== i),
   364→      },
   365→    });
   366→  }
   367→
   368→  return (
   369→    <Card>
   370→      <CardHeader>
   371→        <CardTitle className="font-display">Contact</CardTitle>
   372→        <CardDescription>
   373→          Email, phone, sample address, business hours, and optional map
   374→          embed URL. The address is clearly labeled sample — replace before
   375→          launch.
   376→        </CardDescription>
   377→      </CardHeader>
   378→      <CardContent className="space-y-4">
   379→        <div className="grid gap-3 sm:grid-cols-2">
   380→          <div className="space-y-1.5">
   381→            <Label htmlFor="contact-email" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   382→              Email
   383→            </Label>
   384→            <Input
   385→              id="contact-email"
   386→              type="email"
   387→              value={draft.contact.email}
   388→              onChange={(e) => setDraft({ ...draft, contact: { ...draft.contact, email: e.target.value } })}
   389→            />
   390→          </div>
   391→          <div className="space-y-1.5">
   392→            <Label htmlFor="contact-phone" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   393→              Phone
   394→            </Label>
   395→            <Input
   396→              id="contact-phone"
   397→              type="tel"
   398→              value={draft.contact.phone}
   399→              onChange={(e) => setDraft({ ...draft, contact: { ...draft.contact, phone: e.target.value } })}
   400→            />
   401→          </div>
   402→        </div>
   403→
   404→        <div className="space-y-1.5">
   405→          <Label htmlFor="contact-address" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   406→            Sample address
   407→          </Label>
   408→          <Input
   409→            id="contact-address"
   410→            value={draft.contact.addressSample}
   411→            onChange={(e) => setDraft({ ...draft, contact: { ...draft.contact, addressSample: e.target.value } })}
   412→          />
   413→          <p className="text-[11px]" style={{ color: "var(--text-muted)" }}>
   414→            Sample — replace before launch.
   415→          </p>
   416→        </div>
   417→
   418→        <div className="space-y-1.5">
   419→          <Label htmlFor="contact-map" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   420→            Map embed URL (optional)
   421→          </Label>
   422→          <Input
   423→            id="contact-map"
   424→            value={draft.contact.mapEmbedUrl ?? ""}
   425→            onChange={(e) => setDraft({ ...draft, contact: { ...draft.contact, mapEmbedUrl: e.target.value } })}
   426→            placeholder="https://www.google.com/maps/embed?…"
   427→          />
   428→        </div>
   429→
   430→        <div className="space-y-2">
   431→          <Label className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   432→            Business hours
   433→          </Label>
   434→          <div className="space-y-1.5">
   435→            {draft.contact.hours.map((h, i) => (
   436→              <div key={i} className="flex items-center gap-2">
   437→                <Input
   438→                  value={h.day}
   439→                  onChange={(e) => updateHours(i, { day: e.target.value })}
   440→                  className="h-8 flex-1 text-xs"
   441→                  placeholder="Mon – Fri"
   442→                />
   443→                <Input
   444→                  value={h.hours}
   445→                  onChange={(e) => updateHours(i, { hours: e.target.value })}
   446→                  className="h-8 flex-1 text-xs"
   447→                  placeholder="07:00 – 19:00"
   448→                />
   449→                <Button
   450→                  variant="ghost"
   451→                  size="icon"
   452→                  className="size-8 shrink-0 text-[var(--danger,#ef4444)]"
   453→                  onClick={() => removeHour(i)}
   454→                  aria-label={`Remove hours row ${i + 1}`}
   455→                >
   456→                  <Trash2 className="size-3.5" />
   457→                </Button>
   458→              </div>
   459→            ))}
   460→          </div>
   461→          <Button variant="outline" size="sm" onClick={addHour} className="gap-1.5">
   462→            <Plus className="size-3.5" />
   463→            Add row
   464→          </Button>
   465→        </div>
   466→
   467→        <SaveButton onClick={save} />
   468→      </CardContent>
   469→    </Card>
   470→  );
   471→}

   913→function AdminAppearanceTab() {
   914→  const [appearance, setAppearance] = useState<AdminAppearanceState>(DEFAULT_APPEARANCE);
   915→  const [hydrated, setHydrated] = useState(false);
   916→
   917→  // Hydrate from localStorage
   918→  useEffect(() => {
   919→    try {
   920→      const raw = window.localStorage.getItem(ADMIN_APPEARANCE_KEY);
   921→      if (raw) {
   922→        setAppearance({ ...DEFAULT_APPEARANCE, ...JSON.parse(raw) });
   923→      }
   924→    } catch {
   925→      // ignore
   926→    }
   927→    setHydrated(true);
   928→  }, []);
   929→
   930→  // Persist on change
   931→  useEffect(() => {
   932→    if (!hydrated) return;
   933→    try {
   934→      window.localStorage.setItem(ADMIN_APPEARANCE_KEY, JSON.stringify(appearance));
   935→      // Apply theme to the admin's next-themes key
   936→      window.localStorage.setItem(ADMIN_THEME_KEY, appearance.theme);
   937→      toast.success("Admin appearance saved.");
   938→    } catch {
   939→      // ignore
   940→    }
   941→  }, [appearance, hydrated]);
   942→
   943→  if (!hydrated) {
   944→    return (
   945→      <Card>
   946→        <CardContent className="p-4">
   947→          <p className="text-xs" style={{ color: "var(--text-muted)" }}>
   948→            Loading…
   949→          </p>
   950→        </CardContent>
   951→      </Card>
   952→    );
   953→  }
   954→
   955→  return (
   956→    <Card>
   957→      <CardHeader>
   958→        <CardTitle className="font-display">Admin appearance</CardTitle>
   959→        <CardDescription>
   960→          Theme, sidebar default state, and table density for the admin
   961→          console. Persists to localStorage; applies on next admin visit.

   963→      </CardHeader>

   460→
   461→  function save() {
   462→    updateGlobalSettings(draft);
   463→    toast.success("Contact settings saved.");
   464→  }
   465→
   466→  function updateHours(i: number, next: Partial<{ day: string; hours: string }>) {
   467→    const hours = draft.contact.hours.slice();
   468→    hours[i] = { ...hours[i], ...next };
   469→    setDraft({ ...draft, contact: { ...draft.contact, hours } });
   470→  }
   471→
   472→  function addHour() {
   473→    setDraft({
   474→      ...draft,
   475→      contact: { ...draft.contact, hours: [...draft.contact.hours, { day: "", hours: "" }] },
   476→    });
   477→  }
   478→
   479→  function removeHour(i: number) {
   480→    setDraft({
   481→      ...draft,
   482→      contact: {
   483→        ...draft.contact,
   484→        hours: draft.contact.hours.filter((_, idx) => idx !== i),
   485→      },
   486→    });
   487→  }
   488→
   489→  return (
   490→    <Card>
   491→      <CardHeader>
   492→        <CardTitle className="font-display">Contact</CardTitle>
   493→        <CardDescription>
   494→          Email, phone, sample address, business hours, and optional map

   496→          launch.

   498→      </CardHeader>
   499→      <CardContent className="space-y-4">
   500→        <div className="grid gap-3 sm:grid-cols-2">
   501→          <div className="space-y-1.5">

   504→            </Label>

   506→              id="contact-email"
   507→              type="email"
   508→              value={draft.contact.email}
   509→              onChange={(e) => setDraft({ ...draft, contact: { ...draft.contact, email: e.target.value } })}
   510→            />
   511→          </div>
   512→          <div className="space-y-1.5">
   513→            <Label htmlFor="contact-phone" className="text-xs uppercase tracking-wide" style={{ color: "var(--text-muted)" }}>
   514→              Phone
   515→            </Label>
   516→            <Input
   517→              id="contact-phone"
   518→              type="tel"
   519→              value={draft.contact.phone}
   520→              onChange={(e) => setDraft({ ...draft, contact: { ...draft.contact, phone: e.target.value } })}
   521→            />
   522→          </div>
   523→        </div>
   524→
   525→        <div className="space-y-1.5">

   533→          />

  1156→      if (raw) {
  1157→        setAppearance({ ...DEFAULT_APPEARANCE, ...JSON.parse(raw) });
  1158→      }
  1159→    } catch {
  1160→      // ignore
  1161→    }
  1162→    setHydrated(true);
  1163→  }, []);
  1164→

  1167→    if (!hydrated) return;

  1169→      window.localStorage.setItem(ADMIN_APPEARANCE_KEY, JSON.stringify(appearance));
  1170→      // Apply theme to the admin's next-themes key

  1174→      // ignore
  1175→    }
  1176→  }, [appearance, hydrated]);
  1177→
